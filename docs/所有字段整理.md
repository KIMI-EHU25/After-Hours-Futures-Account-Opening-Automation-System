# 所有字段整理

本文档参考 `service-api/fields.md` 的分组方式，并按照当前后端 POST 请求 Schema 整理。

说明：

- 两个已经停用的只读字段未收入本文档。
- `contactnumber`（联系电话）是当前仍在使用的字段。
- 每个提交接口都需要在请求体中传入 `uid`。
- 日期通过 JSON 传输时使用 `YYYY-MM-DD` 字符串，例如 `2026-03-03`。
- 标记为可选的字段既可以传 `null`，也可以不传。

```js
const data = {
    // p6：反洗钱问卷
    "p6-launderingBack": {
        "uid": 1,
        "p1": "A",      // 6-1：居民税收身份
        "p2": "D",      // 6-2：政要人物关系
                          // 注意：原页面最后两个选项的 E 和 D 的 value 顺序相反
                          // value="E" 为倒数第二个选项，value="D" 为最后一个选项
        "p3": "C",      // 6-3：投资期限
        "p4": "A",      // 6-4：期望收益
        "p5": "A",      // 6-5：账户是否存在被他人控制
        "p6": "A",      // 6-6：账户交易实际受益人为
        "p7": "AB",     // 6-7：想投资的品种，可多选
        "p8": "I",      // 6-8：是否有来自相关机构的不良诚信记录
        "p9": "ABC"     // 6-9：参与期货的主要交易类型，可多选
    },

    // p7：补充资料
    "p7-addInfo": {
        "uid": 1,
        "name": "张三",                                      // 姓名
        "idCardNumber": "110101199003071233",                 // 身份证号
        "idCardAddress": "北京市东城区示例路1号101室",    // 身份证地址
        "idCardValidPeriodStart": "2026-03-03",               // 身份证有效期起始，可选
        "idCardValidPeriodEnd": "2036-03-03",                 // 身份证有效期结束，可选
        "ifIdCardPermanentlyValid": false,                     // 身份证是否长期有效
        "contactAddressProvince": "156310000",                // 联系地址省份代码
        "contactAddressCity": "156310105",                    // 联系地址城市代码
        "contactAddress": "北京市东城区示例路1号101室",     // 联系详细地址
        "postalCode": "200000",                               // 邮政编码
        "departmentProvince": "湖北省",                        // 开户营业部省份
        "departmentCity": "武汉市",                            // 开户营业部城市
        "departmentCategory": "0",                            // 营业部类别
        "departmentName": "2015",                             // 开户营业部
        "contactnumber": "13900000000",                       // 联系电话
        "emailAddress": null,                                  // 电子邮箱，可选
        "occupation": "537",                                  // 职业代码
        "educationalQualification": "3",                      // 学历代码
        "ifIntermediary": false,                               // 是否存在居间人
        "intermediaryName": null,                              // 居间人姓名，可选
        "intermediaryCode": null,                              // 居间人登记编号，可选
        "referencePersonCode": null,                           // 推荐人代码，可选
        "referencePersonName": null                            // 推荐人姓名，可选
    },

    // p10：交易者风险测评问卷
    "p10-question": {
        "uid": 1,
        "p1": "A",      // 10-1：是否具有民事行为能力
        "p2": "A",      // 10-2：主要收入来源
        "p3": "A",      // 10-3：计划用于金融投资的主要资金来源
        "p4": "B",      // 10-4：年平均收入
        "p5": "C",      // 10-5：金融资产情况
        "p6": "A",      // 10-6：是否有尚未清偿的数额较大的债务
        "p7": "A",      // 10-7：投资回报主要用于
        "p8": "B",      // 10-8：专业背景、教育经历、投资知识等
        "p9": "B",      // 10-9：作出金融投资决策的主要依据
        "p10": "B",     // 10-10：投资经历
        "p11": "C",     // 10-11：对交易频率的看法
        "p12": "B",     // 10-12：投资资产分配
        "p13": "A",     // 10-13：可承受的最大损失
        "p14": "C"      // 10-14：杠杆认知/账户亏损情况
    },

    // p12：期货品种选择
    "p12-chooseFutures": {
        "uid": 1,
        "p1": "ABCD"    // 可多选
    }

```

## 接口对应关系

| 数据分组 | 请求接口 |
| --- | --- |
| `p6-launderingBack` | `POST /api/launderingBack/answer` |
| `p7-addInfo` | `POST /api/addInfo/answer` |
| `p10-question` | `POST /api/question/answer` |
| `p12-chooseFutures` | `POST /api/chooseFutures/answer` |

## 可选字段

以下字段可以传 `null` 或不传：

- `idCardValidPeriodStart`
- `idCardValidPeriodEnd`
- `emailAddress`
- `intermediaryName`
- `intermediaryCode`
- `referencePersonCode`
- `referencePersonName`
